POST https://app.osmi-ai.ru/api/v1/prediction/c3ab4cc3-3212-4863-ae99-a3b7e6efd00b                                                                    
                                                                                                                                                    
Заголовки                                    
                                                                                                                                                    
Content-Type: application/json                                                                                                                      
(без авторизации — флоу публичный)           
                                                                                                                                                    
Тело
                                                                                                                                                    
{                                                                                                                                                   
"question": "<base64_PNG>|||<JSON_answers_как_строка>"
}                                                                                                                                                     

Рабочий Python-скрипт (проверен на тестах)                                                                                                            
                                                                                                                                                    
import base64                                                                                                                                         
import json                                                                                                                                         
import requests                              

# 1. Читаем планировку                                                                                                                                
with open('путь/к/плану.jpg', 'rb') as f:
    b64 = base64.b64encode(f.read()).decode()                                                                                                         
                                                                                                                                                    
# 2. Опросник (минимум — style + colors, остальные поля опциональные)                                                                                 
answers = json.dumps({                                                                                                                                
    'style': 'Скандинавский',                                                                                                                         
    'colors': 'Светлые (белый, беж, серый)'                                                                                                         
})                                                                                                                                                    

# 3. Склейка через разделитель |||                                                                                                                    
question = b64 + '|||' + answers                                                                                                                    
                                                                                                                                                    
# 4. Запрос к оркестратору (timeout 600 сек — пайплайн идёт ~2.5 мин)                                                                                 
resp = requests.post(                                                                                                                                 
    'https://app.osmi-ai.ru/api/v1/prediction/c3ab4cc3-3212-4863-ae99-a3b7e6efd00b',                                                                  
    json={'question': question},                                                                                                                      
    timeout=600                              
)                                                                                                                                                     
data = resp.json()                                                                                                                                  
print(data.get('task_id'))
